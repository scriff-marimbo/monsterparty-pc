# RAM / label names for the Monster Party disassembly. Apply with:
#     python3 tools/apply_names.py monsterparty tools/names.txt
# then ./monsterparty/build.sh to confirm the ROM is still byte-identical.
# Safe to re-run; add new lines as you discover things.
#
# RAM:    $ADDR  Name  [size]   ; comment
#         Bytes inside a multi-byte variable become Name+1, Name+2, ...
#         Name may also be an expression (e.g. BossDefeated-1) to alias an address
#         that the code uses as an array base.
#         label OldName Table+1 likewise makes a label an offset into another (its definition line is removed).
# Labels: label OldName NewName ; comment
#         at bankN $ADDR Name ; comment   (new label where there is none yet)

$004C GameMode            ; 00=in game, 08=player died (-> continue screen), 10=entering a door / room change, 40=round complete (boss beaten), 80=title/password entry
$004D GameState           ; 00=title, 02=continue screen, 03-04=pre-round, 05=in game (unconfirmed)
$0051 DoorId              ; door/room id; stable for a given round
$0052 BossRoomFlag        ; bit 7 set = in a boss room (enemy tasks run boss code; a boss hit costs the player 2 health instead of 1)
$0056 BossIdInRound       ; which of the round's bosses this room holds, 1-3 (0 = none). From the door data; its defeated flag is BossDefeated-1+n
$0057 BossId              ; RoundNumber*4 + BossIdInRound (also the boss task's TaskArg0 -> b5_BossHandlersLo/Hi). $0D Giant Cat, $0E Punk Rocker, $15 Chameleon Man, etc.
$005A BossDefeated 3      ; $80 = boss 1/2/3 of this round defeated
$0059 BossDefeated-1      ; code indexes the flags as BossDefeated-1,X with X = 1..3
$03E0 ObjX+10            ; object 10 X (written directly by b4_PlaceObject10)
$03D5 ObjY+10            ; object 10 Y
$008A ScoreDigits 6       ; one decimal digit per byte, tens first (ones digit is always 0 on screen)
$0090 HiScoreDigits 6     ; same layout as ScoreDigits
$0097 RoundNumber         ; 0 = round 1, 1 = round 2, etc
$009E PlayerHealth        ; $2C = full
$0360 TaskCounter          ; per-task field. Bosses: HP (boss = $0366, forms 2/3 = $0367/$0368). Normal enemies: animation step (b5_Enemy_Animate).
                          ; Other tasks reuse it as a counter (e.g. animation step at b5 $86EA).

# Object arrays: 11 entries each, index 0 = player. DrawObject reads them per object.
$03CB ObjY                ; vertical position (copied to sprite Y in DrawObject)
$03D6 ObjX                ; horizontal position (copied to sprite X in DrawObject)
$0423 ObjAttr             ; sprite attribute bits; $40 = flipped horizontally (player: $00 = right, $40 = left)
$03C0 ObjFlags            ; bit 7 = object active (drawn only if this and ObjDrawFlags bit 7 are set); other bits used per object
$0418 ObjDrawFlags        ; bit 7 = visible; low bits pick the graphics set (combined with RoundNumber in DrawObject)
$040D ObjXHi              ; high byte / screen of X; nonzero = off screen, sprites hidden
$042E ObjFrame            ; metasprite / animation frame. Player: 1-2 = bat swing, 3-4 = walking, 6-7 = crawling
